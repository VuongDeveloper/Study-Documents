\chapter{Glossary}
\label{app:glossary}

\newcommand{\gloss}[2]{\item[#1] #2}
\begin{description}\setlength{\itemsep}{2pt}
\gloss{Actuator}{Spring Boot's operational endpoints (\code{/actuator/health},
  \code{/metrics}, \code{/prometheus}); exposure and authorization are
  separate decisions (Ch.\,1).}
\gloss{Advertised listener}{The address a Kafka broker tells clients to
  use after bootstrap --- the usual cause of connects-then-times-out
  (Ch.\,10).}
\gloss{Algorithm confusion}{Attack that rewrites a token's \code{alg}
  header so a verifier applies HMAC with the public key as secret;
  defeated by letting the key type, not the header, choose the
  algorithm (Ch.\,28).}
\gloss{Artifact}{Anything a build produces worth keeping (jar, image,
  report); must be immutable and traceable (Ch.\,21).}
\gloss{Auto-configuration}{Spring Boot conditionally applying
  configuration based on classpath, properties, and user beans (Ch.\,1).}
\gloss{Backend for frontend}{A small service that composes several
  service calls into one client-shaped response; the right home for
  aggregation the gateway must not do (Ch.\,31).}
\gloss{Bootstrap servers}{The initial Kafka addresses a client contacts
  to fetch metadata; all later traffic uses advertised listeners
  (Ch.\,10).}
\gloss{Bulkhead}{Cap on concurrent calls to one dependency (semaphore
  or dedicated thread pool) so a slow dependency cannot consume every
  caller thread (Ch.\,32).}
\gloss{Cipher value}{A \code{\{cipher\}}-prefixed config-server value
  encrypted with the server's key so the repository holds only
  ciphertext; decrypted on serve (Ch.\,29).}
\gloss{Circuit breaker}{State machine over a sliding window of call
  outcomes: CLOSED passes calls, OPEN rejects them instantly for a
  wait period, HALF\_OPEN probes with a few calls (Ch.\,32).}
\gloss{ClusterIP}{Default Service type: a stable virtual IP reachable
  only inside the cluster (Ch.\,18).}
\gloss{Consumer group}{A named team of Kafka consumers splitting a
  topic's partitions; each group keeps its own offsets (Ch.\,10).}
\gloss{Consumer lag}{Distance between a partition's newest offset and a
  group's committed offset; measure it in seconds, alert on it
  (Ch.\,34).}
\gloss{Controller (k8s)}{A reconciliation loop converging actual state
  to desired state for one kind of object (Ch.\,14).}
\gloss{CORS / preflight}{Browser-enforced relaxation of same-origin
  policy; non-simple cross-origin requests cost an extra
  \code{OPTIONS} round trip unless cached --- and never occur when
  page and API share one origin (Ch.\,31, 38).}
\gloss{CQRS}{Command/query separation across services: writes go to the
  owning service, reads to a model built from its events (Ch.\,33).}
\gloss{CrashLoopBackOff}{Pod status: the container keeps exiting;
  restart delay doubles each time (Ch.\,24).}
\gloss{CRD / Operator}{Custom resource kinds plus a controller for them
  --- how Strimzi manages Kafka declaratively (Ch.\,25).}
\gloss{Dead-letter topic}{Where a consumer routes messages that failed
  processing repeatedly (Ch.\,11).}
\gloss{Deployment}{Workload controller for interchangeable replicas;
  manages ReplicaSets; enables rolling updates and rollback (Ch.\,15).}
\gloss{Digest}{The immutable content hash of an image
  (\code{@sha256:...}); tags move, digests do not (Ch.\,13).}
\gloss{Dual write}{Writing to two systems (database and Kafka) with no
  shared transaction; one can succeed while the other fails (Ch.\,33).}
\gloss{Endpoint (k8s)}{A Ready pod backing a Service; the live list is
  maintained by label selector and readiness (Ch.\,18).}
\gloss{Eureka}{Netflix/Spring Cloud service registry: registration,
  heartbeats, client-side caching (Ch.\,3).}
\gloss{Event loop}{Reactor Netty's small fixed pool of threads, each
  multiplexing many sockets with non-blocking I/O; one blocking call
  stalls every request on that loop (Ch.\,31).}
\gloss{explain (Mongo)}{\code{find(...).explain("executionStats")}:
  the query plan with \code{IXSCAN}/\code{FETCH} stages and
  \code{totalDocsExamined} versus \code{nReturned} --- the first thing
  to read when a query is slow (Ch.\,36).}
\gloss{Fallback}{The value or action substituted when a protected call
  fails or is rejected; should be cached, degraded-but-typed, empty,
  or a typed error --- never \code{null} (Ch.\,32).}
\gloss{Flyway}{Versioned, immutable SQL migrations applied at startup
  and recorded in a history table (Ch.\,7).}
\gloss{GitOps}{Deployment model where an in-cluster agent pulls desired
  state from git and converges the cluster to it (Ch.\,26).}
\gloss{Head-based / tail-based sampling}{Deciding at the root span
  whether to keep a trace (cheap, blind to outcome) versus deciding
  after the whole trace is known (keeps errors and slow requests;
  needs a collector) (Ch.\,37).}
\gloss{HS256 / RS256}{JWT signature algorithms. HS256: one shared
  secret both signs and verifies. RS256: RSA private key signs, public
  key verifies --- one issuer, any number of verifiers (Ch.\,28).}
\gloss{HttpOnly cookie}{A cookie JavaScript cannot read; with
  \code{Secure}, \code{SameSite} and a narrow \code{Path} it is the
  safe home for a refresh token (Ch.\,38).}
\gloss{Idempotent consumer}{A listener that claims each event id in the
  same transaction as its side effect, so redelivery is a no-op
  (Ch.\,34).}
\gloss{Idempotent producer}{Kafka producer whose retries the broker
  de-duplicates by sequence number; default since Kafka~3.0
  (Ch.\,34).}
\gloss{Ingress}{L7 HTTP routing object (host/path $\rightarrow$
  Service), executed by an Ingress controller such as Traefik
  (Ch.\,18).}
\gloss{Jib}{Maven plugin building layered OCI images without a Docker
  daemon (Ch.\,13).}
\gloss{JWKS}{JSON Web Key Set: the public keys an issuer publishes at a
  well-known URL so verifiers can fetch and cache them (Ch.\,28).}
\gloss{JWT}{Signed, self-contained identity token:
  header.payload.signature; validated without a database (Ch.\,5;
  at scale, Ch.\,28).}
\gloss{Keyset pagination}{Paging by ``rows after this key'' on an
  indexed sort column instead of \code{skip}, so deep pages cost the
  same as the first (Ch.\,36).}
\gloss{kid}{Key id in a JWT header naming the key that signed it;
  what makes rotation possible while old tokens are in flight
  (Ch.\,28).}
\gloss{KRaft}{Kafka's built-in Raft consensus replacing ZooKeeper
  (Ch.\,10).}
\gloss{kubelet}{The per-node agent that runs pods and executes probes
  (Ch.\,14).}
\gloss{Kustomize}{Template-free manifest composition: bases, overlays,
  patches; built into kubectl (Ch.\,20).}
\gloss{Lease (Eureka)}{An instance's registration, renewed by 30\,s
  heartbeats and expired after 90\,s of silence; swept by the server
  every 60\,s, so a dead instance can linger for minutes (Ch.\,30).}
\gloss{Liveness probe}{``Restart me if this fails'' --- the probe to be
  most conservative with (Ch.\,17).}
\gloss{Multipart upload}{S3 mechanism splitting a large object into
  parts of at least 5\,MB uploaded in parallel and assembled
  server-side; abandoned parts need a lifecycle abort rule (Ch.\,35).}
\gloss{Namespace}{Cluster partition; scope for names and RBAC
  (Ch.\,14).}
\gloss{OOMKilled}{Exit 137: the kernel killed the container for
  exceeding its memory limit; no application trace exists (Ch.\,17).}
\gloss{Outbox pattern}{Making DB-write + event-publish effectively
  atomic by writing the event to a table in the same transaction
  (Ch.\,11; Ch.\,33).}
\gloss{Partition}{Kafka's unit of ordering and parallelism; an ordered
  immutable record sequence (Ch.\,10).}
\gloss{Path-style addressing}{S3 URL form with the bucket in the path
  (\code{host/bucket/key}) rather than as a DNS subdomain; required by
  MinIO without wildcard DNS (Ch.\,35).}
\gloss{Phantom token}{Opaque identifier outside, full JWT inside; the
  gateway swaps one for the other via a fast store, giving thin
  external tokens and instant revocation at the cost of a new
  dependency (Ch.\,28).}
\gloss{Poison pill}{A message that fails deserialization forever,
  stalling its partition unless error handling skips it (Ch.\,11).}
\gloss{Presigned URL}{A time-limited, signed link letting a browser
  access one object directly on S3/MinIO (Ch.\,9).}
\gloss{Probe (startup/readiness/liveness)}{kubelet health checks
  gating restart, traffic, and boot grace respectively (Ch.\,17).}
\gloss{Property source precedence}{Spring Boot's ordered list of
  configuration sources; env vars outrank the config-server document,
  which outranks the packaged YAML (Ch.\,29).}
\gloss{PV / PVC / StorageClass}{Provisioned storage, a workload's
  claim on it, and the factory that makes PVs on demand (Ch.\,19).}
\gloss{RBAC}{Role-based access control: ServiceAccount (identity) +
  Role (verbs on resources) + RoleBinding (the join) (Ch.\,22).}
\gloss{Reconciliation}{The control loop: compare desired and actual
  state, act on the difference, forever (Ch.\,14).}
\gloss{RED / USE}{Rate, Errors, Duration for request-driven services;
  Utilization, Saturation, Errors for resources --- the two dashboard
  vocabularies (Ch.\,37).}
\gloss{Refresh scope}{Beans rebuilt on \code{/actuator/refresh} so a
  config change applies without restart; per instance unless
  broadcast by Spring Cloud Bus (Ch.\,29).}
\gloss{Registry}{Image storage speaking the Docker/OCI protocol; the
  interface between builder and cluster (Ch.\,13).}
\gloss{Relaxed binding}{Spring's mapping of
  \code{SPRING\_DATASOURCE\_URL} to \code{spring.datasource.url}
  (Ch.\,1).}
\gloss{ReplicaSet}{The counter keeping N pods of one template alive;
  managed by Deployments, kept for rollback (Ch.\,15).}
\gloss{Retry storm}{Synchronized retries from many clients that
  multiply load on an already failing dependency; prevented by jitter,
  retry budgets, and retrying only idempotent calls (Ch.\,32).}
\gloss{Retry topic}{Per-delay topic a failed record is re-published to
  (\code{-retry-2000}, \ldots), ending in a dead-letter topic; Spring's
  \code{@RetryableTopic} (Ch.\,34).}
\gloss{Rolling update}{Replacing pods incrementally, gated on
  readiness, so service never fully stops (Ch.\,15).}
\gloss{Saga}{A multi-service write as a chain of local transactions,
  each with a compensating action if a later step fails (Ch.\,33).}
\gloss{Secret (k8s)}{Namespaced key/value object for credentials;
  base64-encoded, protected by RBAC not cryptography (Ch.\,16).}
\gloss{Self-preservation}{Eureka mode entered when renewals fall
  below 85\,\% of expected; eviction stops entirely --- triggered by a
  single crash on a five-instance fleet (Ch.\,30).}
\gloss{Service (k8s)}{Stable name + virtual IP over a label-selected,
  readiness-filtered set of pods (Ch.\,18).}
\gloss{ServiceAccount}{A non-human identity for API access --- what
  Jenkins deploys as (Ch.\,22).}
\gloss{SigV4}{AWS Signature Version 4: an HMAC chain over date, region
  and service signs a canonical request that always includes the
  \code{Host} header --- a presigned URL is bound to the host it was
  signed for (Ch.\,35).}
\gloss{Single-flight refresh}{One shared pending promise so that
  concurrent 401s trigger exactly one token refresh and all replay
  with its result; mandatory once refresh tokens rotate (Ch.\,38).}
\gloss{StatefulSet}{Workload controller for stateful replicas: stable
  names, per-replica PVCs, ordered replacement (Ch.\,15).}
\gloss{Text index (Mongo)}{Index that tokenizes and stems string
  fields with weights, queried via \code{\$text}/\code{TextCriteria};
  matches whole terms, not substrings (Ch.\,36).}
\gloss{Token bucket}{Rate-limit algorithm: tokens refill at a steady
  rate up to a burst capacity and each request spends one; Spring
  Cloud Gateway keeps the bucket in Redis (Ch.\,31).}
\gloss{TOTP}{Time-based one-time passwords: shared secret + clock,
  RFC 6238 (Ch.\,5).}
\gloss{traceparent}{The W3C Trace Context header carrying trace id,
  parent span id and sampling flag between services; Boot~3 emits it
  and also accepts B3 (Ch.\,37).}
\gloss{Trusted edge}{Gateway validates JWTs once and forwards identity
  headers; services trust them --- valid only if no path bypasses the
  gateway (Ch.\,5).}
\end{description}
